{$CLEO .cs}

// GTA San Andreas - Mundo Expandido
// Interfaz base y panel de administrador. Objetivo: GTA SA 1.0 US.

0000:
03A4: name_thread 'ME_CORE'
0@ = 0
1@ = 0

:ME_MAIN
wait 0

// F5: abrir/cerrar panel principal.
if
    0AB0: key_pressed 0x74
then
    if
        0@ == 0
    then
        0@ = 1
        0ACD: show_text_highpriority "MUNDO EXPANDIDO~n~~w~Panel principal~n~~y~F5: cerrar~n~F6: administrador" time 4000
    else
        0@ = 0
        0ACD: show_text_highpriority "Panel principal cerrado" time 1500
    end
    :ME_WAIT_F5
    wait 0
    if
        0AB0: key_pressed 0x74
    then
        jump @ME_WAIT_F5
    end
end

// F6: abrir/cerrar panel de administrador.
if
    0AB0: key_pressed 0x75
then
    if
        1@ == 0
    then
        1@ = 1
        0ACD: show_text_highpriority "~r~ADMINISTRADOR~n~~w~1: +$10,000~n~2: +$100,000~n~3: Salud y chaleco~n~4: Quitar estrellas~n~5: +$1,000,000~n~F6: cerrar" time 5000
    else
        1@ = 0
        0ACD: show_text_highpriority "Panel de administrador cerrado" time 1500
    end
    :ME_WAIT_F6
    wait 0
    if
        0AB0: key_pressed 0x75
    then
        jump @ME_WAIT_F6
    end
end

// Acciones administrativas: solo funcionan con el panel abierto.
if
    1@ == 1
then
    // 1: +$10,000
    if
        0AB0: key_pressed 0x31
    then
        0109: player $PLAYER_CHAR money += 10000
        0ACD: show_text_highpriority "~g~Administrador: +$10,000" time 1500
        :ME_WAIT_1
        wait 0
        if
            0AB0: key_pressed 0x31
        then
            jump @ME_WAIT_1
        end
    end

    // 2: +$100,000
    if
        0AB0: key_pressed 0x32
    then
        0109: player $PLAYER_CHAR money += 100000
        0ACD: show_text_highpriority "~g~Administrador: +$100,000" time 1500
        :ME_WAIT_2
        wait 0
        if
            0AB0: key_pressed 0x32
        then
            jump @ME_WAIT_2
        end
    end

    // 3: restaurar salud y chaleco.
    if
        0AB0: key_pressed 0x33
    then
        0223: set_actor $PLAYER_ACTOR health_to 100
        035F: set_actor $PLAYER_ACTOR armour_to 100
        0ACD: show_text_highpriority "~g~Salud y chaleco restaurados" time 1500
        :ME_WAIT_3
        wait 0
        if
            0AB0: key_pressed 0x33
        then
            jump @ME_WAIT_3
        end
    end

    // 4: eliminar nivel de busqueda policial.
    if
        0AB0: key_pressed 0x34
    then
        010D: set_player $PLAYER_CHAR wanted_level_to 0
        0110: clear_player $PLAYER_CHAR wanted_level
        0ACD: show_text_highpriority "~g~Nivel de busqueda eliminado" time 1500
        :ME_WAIT_4
        wait 0
        if
            0AB0: key_pressed 0x34
        then
            jump @ME_WAIT_4
        end
    end

    // 5: +$1,000,000
    if
        0AB0: key_pressed 0x35
    then
        0109: player $PLAYER_CHAR money += 1000000
        0ACD: show_text_highpriority "~g~Administrador: +$1,000,000" time 1500
        :ME_WAIT_5
        wait 0
        if
            0AB0: key_pressed 0x35
        then
            jump @ME_WAIT_5
        end
    end
end

// ESC: cerrar ambos paneles.
if
    0AB0: key_pressed 0x1B
then
    0@ = 0
    1@ = 0
    0ACD: show_text_highpriority "Paneles cerrados" time 1200
    :ME_WAIT_ESC
    wait 0
    if
        0AB0: key_pressed 0x1B
    then
        jump @ME_WAIT_ESC
    end
end

jump @ME_MAIN
