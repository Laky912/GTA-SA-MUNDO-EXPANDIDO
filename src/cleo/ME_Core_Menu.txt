{$CLEO .cs}

// GTA San Andreas - Mundo Expandido
// Modulo inicial: menu de juego y panel de administrador.
// Objetivo: GTA San Andreas clasico PC, version 1.0 US.

0000:
03A4: name_thread 'ME_CORE'
0@ = 0 // Menu principal: 0 cerrado, 1 abierto.
1@ = 0 // Panel admin: 0 cerrado, 1 abierto.
10@ = 0 // Estado del trabajo de reparto.
11@ = 0.0 // Distancia acumulada del reparto.
19@ = 0 // Avisos de progreso del reparto.
20@ = 0 // Modo elegido en el selector del menu inicial.
0AF0: 20@ = get_int_from_ini_file "CLEO\\ME_MODE.ini" section "mode" key "selected"

:ME_MAIN
wait 0

// F5 abre/cierra el menu funcional de Mundo Expandido.
if
    0AB0: key_pressed 0x74
then
    if
        0@ == 0
    then
        0@ = 1
        1@ = 0
        0ACD: show_text_highpriority "~b~MUNDO EXPANDIDO~n~~w~1: Recuperar salud~n~2: Reparar chaleco~n~3: Quitar estrellas~n~4: Recibir $5,000~n~F7: trabajo de reparto~n~F5: cerrar | F6: admin" time 6000
    else
        0@ = 0
        0ACD: show_text_highpriority "Menu principal cerrado" time 1500
    end
    :ME_WAIT_F5
    wait 0
    if
        0AB0: key_pressed 0x74
    then
        jump @ME_WAIT_F5
    end
end

// Acciones del menu principal.
if
    0@ == 1
then
    if
        0AB0: key_pressed 0x31
    then
        0223: set_actor $PLAYER_ACTOR health_to 100
        0ACD: show_text_highpriority "~g~Salud recuperada" time 1500
        :ME_MENU_WAIT_1
        wait 0
        if
            0AB0: key_pressed 0x31
        then
            jump @ME_MENU_WAIT_1
        end
    end

    if
        0AB0: key_pressed 0x32
    then
        035F: set_actor $PLAYER_ACTOR armour_to 100
        0ACD: show_text_highpriority "~g~Chaleco restaurado" time 1500
        :ME_MENU_WAIT_2
        wait 0
        if
            0AB0: key_pressed 0x32
        then
            jump @ME_MENU_WAIT_2
        end
    end

    if
        0AB0: key_pressed 0x33
    then
        010D: set_player $PLAYER_CHAR wanted_level_to 0
        0110: clear_player $PLAYER_CHAR wanted_level
        0ACD: show_text_highpriority "~g~Busqueda policial eliminada" time 1500
        :ME_MENU_WAIT_3
        wait 0
        if
            0AB0: key_pressed 0x33
        then
            jump @ME_MENU_WAIT_3
        end
    end

    if
        0AB0: key_pressed 0x34
    then
        0109: player $PLAYER_CHAR money += 5000
        0ACD: show_text_highpriority "~g~Has recibido $5,000" time 1500
        :ME_MENU_WAIT_4
        wait 0
        if
            0AB0: key_pressed 0x34
        then
            jump @ME_MENU_WAIT_4
        end
    end
end

// F7 inicia o cancela un trabajo de reparto.
// Requiere conducir; al recorrer 750 metros se pagan $2,500.
if
    0AB0: key_pressed 0x76
then
    if
        10@ == 0
    then
        if
            00DF: actor $PLAYER_ACTOR driving
        then
            10@ = 1
            11@ = 0.0
            19@ = 0
            00A0: store_actor $PLAYER_ACTOR position_to 12@ 13@ 14@
            0ACD: show_text_highpriority "~b~REPARTO INICIADO~n~~w~Conduce 750 metros para cobrar $2,500" time 4000
        else
            0ACD: show_text_highpriority "~r~Debes estar dentro de un vehiculo" time 2500
        end
    else
        10@ = 0
        0ACD: show_text_highpriority "Trabajo de reparto cancelado" time 2000
    end
    :ME_WAIT_F7
    wait 0
    if
        0AB0: key_pressed 0x76
    then
        jump @ME_WAIT_F7
    end
end

// Seguimiento de distancia del trabajo activo.
if
    10@ == 1
then
    00A0: store_actor $PLAYER_ACTOR position_to 15@ 16@ 17@
    if
        00DF: actor $PLAYER_ACTOR driving
    then
        050A: get_distance_between_coords_2d 15@ 16@ 12@ 13@ store_to 18@
        // Ignora saltos imposibles entre fotogramas para evitar completar
        // el trabajo mediante teletransportes o cambios bruscos de posicion.
        if
            18@ < 100.0
        then
            005B: 11@ += 18@
        end
        if
            and
                11@ >= 250.0
                19@ == 0
        then
            19@ = 1
            0ACD: show_text_highpriority "~y~Reparto: 250 / 750 metros" time 1800
        end
        if
            and
                11@ >= 500.0
                19@ == 1
        then
            19@ = 2
            0ACD: show_text_highpriority "~y~Reparto: 500 / 750 metros" time 1800
        end
        if
            11@ >= 750.0
        then
            10@ = 0
            0109: player $PLAYER_CHAR money += 2500
            0ACD: show_text_highpriority "~g~Reparto completado~n~~w~Has ganado $2,500" time 4000
        end
    end
    0087: 12@ = 15@
    0087: 13@ = 16@
end

// F6 abre/cierra el panel de administrador.
// Si el modo seleccionado no es administrador, F6 no abre el panel.
if
    and
        0AB0: key_pressed 0x75
        20@ <> 2
then
    0ACD: show_text_highpriority "~r~El panel admin solo esta disponible en Libre + Admin" time 2500
    :ME_WAIT_F6_LOCKED
    wait 0
    if
        0AB0: key_pressed 0x75
    then
        jump @ME_WAIT_F6_LOCKED
    end
end

if
    and
        0AB0: key_pressed 0x75
        20@ == 2
then
    if
        1@ == 0
    then
        1@ = 1
        0@ = 0
        0ACD: show_text_highpriority "~r~ADMINISTRADOR~n~~w~1: +$10,000~n~2: +$100,000~n~3: Salud y chaleco~n~4: Quitar estrellas~n~5: +$1,000,000~n~F6: cerrar" time 6000
    else
        1@ = 0
        0ACD: show_text_highpriority "Panel de administrador cerrado" time 1500
    end
    :ME_WAIT_F6
    wait 0
    if
        0AB0: key_pressed 0x75
    then
        jump @ME_WAIT_F6
    end
end

// Acciones administrativas: solo funcionan con el panel abierto.
if
    1@ == 1
then
    if
        0AB0: key_pressed 0x31
    then
        0109: player $PLAYER_CHAR money += 10000
        0ACD: show_text_highpriority "~g~Administrador: +$10,000" time 1500
        :ME_WAIT_1
        wait 0
        if
            0AB0: key_pressed 0x31
        then
            jump @ME_WAIT_1
        end
    end

    if
        0AB0: key_pressed 0x32
    then
        0109: player $PLAYER_CHAR money += 100000
        0ACD: show_text_highpriority "~g~Administrador: +$100,000" time 1500
        :ME_WAIT_2
        wait 0
        if
            0AB0: key_pressed 0x32
        then
            jump @ME_WAIT_2
        end
    end

    if
        0AB0: key_pressed 0x33
    then
        0223: set_actor $PLAYER_ACTOR health_to 100
        035F: set_actor $PLAYER_ACTOR armour_to 100
        0ACD: show_text_highpriority "~g~Salud y chaleco restaurados" time 1500
        :ME_WAIT_3
        wait 0
        if
            0AB0: key_pressed 0x33
        then
            jump @ME_WAIT_3
        end
    end

    if
        0AB0: key_pressed 0x34
    then
        010D: set_player $PLAYER_CHAR wanted_level_to 0
        0110: clear_player $PLAYER_CHAR wanted_level
        0ACD: show_text_highpriority "~g~Nivel de busqueda eliminado" time 1500
        :ME_WAIT_4
        wait 0
        if
            0AB0: key_pressed 0x34
        then
            jump @ME_WAIT_4
        end
    end

    if
        0AB0: key_pressed 0x35
    then
        0109: player $PLAYER_CHAR money += 1000000
        0ACD: show_text_highpriority "~g~Administrador: +$1,000,000" time 1500
        :ME_WAIT_5
        wait 0
        if
            0AB0: key_pressed 0x35
        then
            jump @ME_WAIT_5
        end
    end
end

// ESC cierra ambos paneles.
if
    0AB0: key_pressed 0x1B
then
    0@ = 0
    1@ = 0
    0ACD: show_text_highpriority "Paneles cerrados" time 1200
    :ME_WAIT_ESC
    wait 0
    if
        0AB0: key_pressed 0x1B
    then
        jump @ME_WAIT_ESC
    end
end

jump @ME_MAIN
