{$CLEO .cs}

// GTA San Andreas - Mundo Expandido
// Core de interfaz y panel de administrador para GTA SA 1.0 US.
// Fuente Sanny Builder. Compilar antes de instalar.

0000:
03A4: name_thread 'ME_CORE'
0@ = 0
1@ = 0

:ME_MAIN
wait 0

// Panel principal: F5
if
    0AB0: key_pressed 0x74
then
    wait 250
    if
        0AB0: key_pressed 0x74
    then
        if
            0@ == 0
        then
            0@ = 1
            0ACD: show_text_highpriority "MUNDO EXPANDIDO~n~~w~Panel principal~n~~y~F5: cerrar panel~n~F6: administrador" time 5000
        else
            0@ = 0
            0ACD: show_text_highpriority "Panel principal cerrado" time 1500
        end
        :ME_WAIT_F5
        wait 0
        if
            0AB0: key_pressed 0x74
        then
            jump @ME_WAIT_F5
        end
    end
end

// Panel de administrador: F6. Independiente del panel principal.
if
    0AB0: key_pressed 0x75
then
    wait 250
    if
        0AB0: key_pressed 0x75
    then
        if
            1@ == 0
        then
            1@ = 1
            0ACD: show_text_highpriority "~r~ADMINISTRADOR~n~~w~1: +$10,000~n~2: +$100,000~n~3: Salud y chaleco~n~4: Quitar estrellas~n~F6: cerrar" time 5000
        else
            1@ = 0
            0ACD: show_text_highpriority "Panel de administrador cerrado" time 1500
        end
        :ME_WAIT_F6
        wait 0
        if
            0AB0: key_pressed 0x75
        then
            jump @ME_WAIT_F6
        end
    end
end

// Acciones disponibles solo mientras el panel de administrador esta abierto.
if
    1@ == 1
then
    // 1: agregar diez mil dolares
    if
        0AB0: key_pressed 0x31
    then
        wait 200
        if
            0AB0: key_pressed 0x31
        then
            0109: player $PLAYER_CHAR money += 10000
            0ACD: show_text_highpriority "~g~Administrador: +$10,000" time 2000
            :ME_WAIT_1
            wait 0
            if
                0AB0: key_pressed 0x31
            then
                jump @ME_WAIT_1
            end
        end
    end

    // 2: agregar cien mil dolares
    if
        0AB0: key_pressed 0x32
    then
        wait 200
        if
            0AB0: key_pressed 0x32
        then
            0109: player $PLAYER_CHAR money += 100000
            0ACD: show_text_highpriority "~g~Administrador: +$100,000" time 2000
            :ME_WAIT_2
            wait 0
            if
                0AB0: key_pressed 0x32
            then
                jump @ME_WAIT_2
            end
        end
    end

    // 3: restaurar salud y chaleco
    if
        0AB0: key_pressed 0x33
    then
        wait 200
        if
            0AB0: key_pressed 0x33
        then
            0223: set_actor $PLAYER_ACTOR health_to 100
            035F: set_actor $PLAYER_ACTOR armour_to 100
            0ACD: show_text_highpriority "~g~Salud y chaleco restaurados" time 2000
            :ME_WAIT_3
            wait 0
            if
                0AB0: key_pressed 0x33
            then
                jump @ME_WAIT_3
            end
        end
    end

    // 4: eliminar nivel de busqueda policial
    if
        0AB0: key_pressed 0x34
    then
        wait 200
        if
            0AB0: key_pressed 0x34
        then
            010D: set_player $PLAYER_CHAR wanted_level_to 0
            0110: clear_player $PLAYER_CHAR wanted_level
            0ACD: show_text_highpriority "~g~Nivel de busqueda eliminado" time 2000
            :ME_WAIT_4
            wait 0
            if
                0AB0: key_pressed 0x34
            then
                jump @ME_WAIT_4
            end
        end
    end
end

// ESC cierra ambos paneles.
if
    0AB0: key_pressed 0x1B
then
    wait 250
    if
        0AB0: key_pressed 0x1B
    then
        0@ = 0
        1@ = 0
        0ACD: show_text_highpriority "Paneles cerrados" time 1500
        :ME_WAIT_ESC
        wait 0
        if
            0AB0: key_pressed 0x1B
        then
            jump @ME_WAIT_ESC
        end
    end
end

jump @ME_MAIN
