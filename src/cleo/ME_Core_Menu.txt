{$CLEO .cs}

// GTA San Andreas - Mundo Expandido
// Base de interfaz F5 para SA 1.0 US.
// Fuente Sanny Builder: compilar antes de instalar.

0000:
03A4: name_thread 'ME_CORE'
0@ = 0

:ME_MAIN
wait 0

// F5 alterna el panel. Esperar a soltar la tecla evita repeticiones.
if
    0AB0: key_pressed 0x74
then
    wait 250
    if
        0AB0: key_pressed 0x74
    then
        if
            0@ == 0
        then
            0@ = 1
            0ACD: show_text_highpriority "MUNDO EXPANDIDO~n~~w~Panel principal~n~~y~F5: cerrar panel~n~ESC: cerrar panel" time 5000
        else
            0@ = 0
            0ACD: show_text_highpriority "Panel cerrado" time 1500
        end

        :ME_WAIT_F5_RELEASE
        wait 0
        if
            0AB0: key_pressed 0x74
        then
            jump @ME_WAIT_F5_RELEASE
        end
    end
end

// ESC cierra el panel cuando está abierto.
if
    0@ == 1
then
    if
        0AB0: key_pressed 0x1B
    then
        wait 250
        if
            0AB0: key_pressed 0x1B
        then
            0@ = 0
            0ACD: show_text_highpriority "Panel cerrado" time 1500
            :ME_WAIT_ESC_RELEASE
            wait 0
            if
                0AB0: key_pressed 0x1B
            then
                jump @ME_WAIT_ESC_RELEASE
            end
        end
    end
end

jump @ME_MAIN
